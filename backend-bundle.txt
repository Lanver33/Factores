"""
Pipeline de Inversión y Reconciliación Contable: Cuentas Públicas de España (SEC 2010 / ESA 2010).
Consulta la API v1.0 de Eurostat (datasets gov_10a_main, gov_10a_taxag, gov_10a_exp) para el sector S.13 (AAPP),
reconcilia identidades fiscales, aplica taxonomías COFOG/tributarias y genera los artefactos de datos.
"""

from datetime import date
import json
import math
import os
from pathlib import Path
import sys
from typing import Optional

from eurostat_client import fetch_eurostat_dataset, EurostatDataset
from utils import publish_dataset


def load_config(config_path: Path) -> dict:
    with open(config_path, "r", encoding="utf-8") as f:
        return json.load(f)


def build_cuentas_publicas_payload(
    config: dict,
    ds_main: EurostatDataset,
    ds_tax: EurostatDataset,
    ds_exp: EurostatDataset,
) -> dict:
    """
    Procesa, pivota y reconcilia los 3 datasets de Eurostat aplicando las reglas contables SEC 2010.
    """
    start_year = config.get("startYear", 1995)

    main_lookup = {}
    for r in ds_main.records:
        if r.get("geo") == config["geo"] and r.get("sector") == config["sector"]:
            unit = r.get("unit")
            time_per = r.get("time")
            na_item = r.get("na_item")
            if unit and time_per and na_item:
                main_lookup[(unit, time_per, na_item)] = r["value"]

    tax_lookup = {}
    for r in ds_tax.records:
        if r.get("geo") == config["geo"] and r.get("sector") == config["sector"]:
            unit = r.get("unit")
            time_per = r.get("time")
            na_item = r.get("na_item")
            if unit and time_per and na_item:
                tax_lookup[(unit, time_per, na_item)] = r["value"]

    exp_lookup = {}
    for r in ds_exp.records:
        if r.get("geo") == config["geo"] and r.get("sector") == config["sector"]:
            unit = r.get("unit")
            time_per = r.get("time")
            cofog = r.get("cofog99")
            if unit and time_per and cofog:
                exp_lookup[(unit, time_per, cofog)] = r["value"]

    units = config.get("units", [])
    unit_ids = [u["id"] for u in units]

    all_time_periods = set()
    for (u, t, item), _ in main_lookup.items():
        if int(t) >= start_year and item in ("TR", "TE", "B9"):
            all_time_periods.add(t)

    valid_years = []
    for y in sorted(all_time_periods):
        complete_for_year = True
        for uid in unit_ids:
            if (uid, y, "TR") not in main_lookup or (uid, y, "TE") not in main_lookup or (uid, y, "B9") not in main_lookup:
                complete_for_year = False
                break
        if complete_for_year:
            valid_years.append(y)

    if len(valid_years) < 10:
        raise RuntimeError(f"Se encontraron solo {len(valid_years)} años válidos (se requieren al menos 10 desde {start_year}).")

    min_year = valid_years[0]
    max_year = valid_years[-1]

    cofog_years = set()
    divs_10 = [f"GF0{i}" if i < 10 else "GF10" for i in range(1, 11)]
    for y in valid_years:
        has_all_cofog = True
        for uid in unit_ids:
            for div in divs_10:
                if (uid, y, div) not in exp_lookup:
                    has_all_cofog = False
                    break
            if not has_all_cofog:
                break
        if has_all_cofog:
            cofog_years.add(y)

    min_cofog_year = min(cofog_years) if cofog_years else min_year
    max_cofog_year = max(cofog_years) if cofog_years else min_year

    annual_balances = []
    top_level_datapoints = []

    decimals_map = {u["id"]: u.get("decimals", 2) for u in units}

    for y in valid_years:
        decimals_mio = decimals_map.get("MIO_EUR", 1)
        decimals_pct = decimals_map.get("PC_GDP", 2)

        cofog_available = y in cofog_years
        if not cofog_available:
            print(f"      ⚠️ Advertencia: Datos COFOG aún no publicados para el año {y}; marcando cofogAvailable=false.")

        metrics_dict = {}

        for uid in unit_ids:
            dec = decimals_map.get(uid, 2)
            tr = main_lookup.get((uid, y, "TR"), 0.0)
            te = main_lookup.get((uid, y, "TE"), 0.0)
            b9 = main_lookup.get((uid, y, "B9"), 0.0)

            diff_b9 = (tr - te) - b9
            tol_b9 = 1.0 if uid == "MIO_EUR" else 0.2
            if abs(diff_b9) > tol_b9:
                raise ValueError(f"Fallo Identidad 1 (Balance Fiscal) en {y} [{uid}]: (TR {tr} - TE {te}) - B9 {b9} = {diff_b9} (tol={tol_b9})")

            d2 = tax_lookup.get((uid, y, "D2"), 0.0)
            d5 = tax_lookup.get((uid, y, "D5"), 0.0)
            d61 = tax_lookup.get((uid, y, "D61"), 0.0)
            d91 = tax_lookup.get((uid, y, "D91"), 0.0)
            d995 = tax_lookup.get((uid, y, "D995"), 0.0)
            d_net_agg = tax_lookup.get((uid, y, "D2_D5_D91_D61_M_D995"))

            tax_gross = d2 + d5 + d61 + d91
            uncollectible_adj = 0.0
            total_tax_net = tax_gross

            if d_net_agg is not None:
                total_tax_net = d_net_agg
                uncollectible_adj = total_tax_net - tax_gross
            elif d995 != 0.0:
                total_tax_net = tax_gross - abs(d995)
                uncollectible_adj = -abs(d995)

            non_tax_residual = tr - total_tax_net

            diff_tax = (tax_gross + uncollectible_adj + non_tax_residual) - tr
            tol_tax = 1.0 if uid == "MIO_EUR" else 0.2
            if abs(diff_tax) > tol_tax:
                raise ValueError(f"Fallo Identidad 2 (Completitud de Ingresos) en {y} [{uid}]: diff={diff_tax} (tol={tol_tax})")

            tax_code_mapping = {
                "D2": ["D211", "D214", "D212", "D29"],
                "D5": ["D51A_C1", "D51B_C2", "D59"],
                "D61": ["D611", "D613", "D612"],
                "D91": [],
            }

            revenues_block = {}
            for tcode, children in tax_code_mapping.items():
                tval = tax_lookup.get((uid, y, tcode))
                if tval is not None and math.isfinite(tval):
                    breakdown = {}
                    for ccode in children:
                        cval = tax_lookup.get((uid, y, ccode))
                        if cval is not None and math.isfinite(cval):
                            breakdown[ccode] = round(cval, dec)
                    revenues_block[tcode] = {
                        "total": round(tval, dec),
                        "breakdown": breakdown,
                    }

            revenues_block["NON_TAX_RESIDUAL"] = {
                "total": round(non_tax_residual, dec),
                "breakdown": {},
            }

            expenditures_block = None
            if cofog_available:
                expenditures_block = {}
                cofog_mapping = {
                    "GF10": ["GF1002", "GF1005"],
                    "GF07": [],
                    "GF01": ["GF0107", "GF0101"],
                    "GF09": [],
                    "GF04": ["GF0405"],
                    "GF03": ["GF0301", "GF0303"],
                    "GF08": [],
                    "GF02": [],
                    "GF05": [],
                    "GF06": [],
                }
                sum_cofog = 0.0
                for fcode, highlights in cofog_mapping.items():
                    fval = exp_lookup.get((uid, y, fcode))
                    if fval is not None and math.isfinite(fval):
                        sum_cofog += fval
                        h_dict = {}
                        for hcode in highlights:
                            hval = exp_lookup.get((uid, y, hcode))
                            if hval is not None and math.isfinite(hval):
                                h_dict[hcode] = round(hval, dec)
                        expenditures_block[fcode] = {
                            "total": round(fval, dec),
                            "highlights": h_dict,
                        }

                total_node = exp_lookup.get((uid, y, "TOTAL"))
                tol_cofog = 1.5 if uid == "MIO_EUR" else 0.25
                if total_node is not None and math.isfinite(total_node):
                    if abs(sum_cofog - total_node) > tol_cofog:
                        raise ValueError(f"Fallo Identidad 3a (Suma GF01..GF10 vs TOTAL) en {y} [{uid}]: sum {sum_cofog} != TOTAL {total_node}")
                    tol_te = 250.0 if uid == "MIO_EUR" else 0.5
                    if abs(total_node - te) > tol_te:
                        raise ValueError(f"Fallo Identidad 3b (TOTAL vs TE) en {y} [{uid}]: TOTAL {total_node} != TE {te} (diff={abs(total_node - te)})")

            totals_dict = {
                "revenue": round(tr, dec),
                "expenditure": round(te, dec),
                "balance": round(b9, dec),
            }
            if abs(uncollectible_adj) > 0.001:
                totals_dict["uncollectibleAdjustment"] = round(uncollectible_adj, dec)

            metrics_dict[uid] = {
                "totals": totals_dict,
                "revenues": revenues_block,
                "expenditures": expenditures_block,
            }

        annual_balances.append({
            "year": y,
            "cofogAvailable": cofog_available,
            "metrics": metrics_dict,
        })

        pc_gdp_b9 = main_lookup.get(("PC_GDP", y, "B9"))
        if pc_gdp_b9 is None or not math.isfinite(pc_gdp_b9):
            raise ValueError(f"Identidad 4: Valor B9 nulo o no finito en PC_GDP para el año {y}.")
        top_level_datapoints.append({
            "year": y,
            "value": round(pc_gdp_b9, decimals_pct),
        })

    output_sources = []
    for src in config.get("sources", []):
        dkey = src.get("datasetKey")
        s_copy = {
            "institution": src["institution"],
            "shortName": src["shortName"],
            "title": src["title"],
            "url": src["url"],
        }
        if "tableId" in src:
            s_copy["tableId"] = src["tableId"]

        if dkey in ("aggregates", "revenues"):
            s_copy["period"] = f"{min_year}-{max_year}"
            s_copy["lastUpdated"] = ds_main.updated or date.today().isoformat()
        elif dkey == "expenditures":
            s_copy["period"] = f"{min_cofog_year}-{max_cofog_year}"
            s_copy["lastUpdated"] = ds_exp.updated or date.today().isoformat()
        elif dkey is None:
            s_copy["period"] = f"{min_year}-{max_year}"
            latest_upd = max(
                filter(None, [ds_main.updated, ds_tax.updated, ds_exp.updated]),
                default=date.today().isoformat()
            )
            s_copy["lastUpdated"] = latest_upd

        output_sources.append(s_copy)

    latest_upd_global = max(
        filter(None, [ds_main.updated, ds_tax.updated, ds_exp.updated]),
        default=date.today().isoformat()
    )

    payload_out = {
        "id": config["indicatorId"],
        "kind": config["chartType"],
        "category": config["category"],
        "valueFormat": "percent",
        "lastUpdated": latest_upd_global,
        "availableUnits": config["units"],
        "taxonomy": config["taxonomy"],
        "dataPoints": top_level_datapoints,
        "annualBalances": annual_balances,
        "contextNotes": [
            "Serie histórica homogénea bajo la metodología del Sistema Europeo de Cuentas (SEC 2010 / ESA 2010) para el sector consolidado de las Administraciones Públicas de España (S.13: Administración Central, Comunidades Autónomas, Corporaciones Locales y Seguridad Social).",
            "El saldo fiscal (B.9, capacidad o necesidad de financiación) equivale a Total Ingresos (TR) menos Total Gastos (TE). Un signo negativo representa déficit público y un signo positivo representa superávit.",
            "El desglose funcional del gasto sigue la clasificación internacional COFOG 1999 (divisiones GF01 a GF10). Los intereses de la deuda pública se contabilizan dentro de Servicios Públicos Generales (subfunción GF0107).",
            "Los ingresos tributarios proceden de los agregados fiscales SEC 2010 (D.2, D.5, D.61 y D.91). Los ingresos no tributarios se obtienen por diferencia frente a TR e incluyen tasas, ventas de mercado, rentas de la propiedad y transferencias corrientes y de capital de la Unión Europea."
        ],
        "sources": output_sources,
    }

    return payload_out


def run_pipeline(config_path: Path, output_path: Path, dry_run: bool = False, guide_path: Optional[Path] = None) -> dict:
    config = load_config(config_path)
    base_url = config["apiBaseUrl"]
    datasets_cfg = config["datasets"]

    print(f"\n[1/3] 📥 Descargando datasets de Eurostat para {config['indicatorId']}...")
    ds_main = fetch_eurostat_dataset(base_url, datasets_cfg["aggregates"]["code"], datasets_cfg["aggregates"]["params"], since_year=config["startYear"])
    ds_tax = fetch_eurostat_dataset(base_url, datasets_cfg["revenues"]["code"], datasets_cfg["revenues"]["params"], since_year=config["startYear"])
    ds_exp = fetch_eurostat_dataset(base_url, datasets_cfg["expenditures"]["code"], datasets_cfg["expenditures"]["params"], since_year=config["startYear"])

    print("\n[2/3] ⚖️ Procesando, reconciliando y aplicando identidades contables SEC 2010...")
    payload = build_cuentas_publicas_payload(config, ds_main, ds_tax, ds_exp)

    if guide_path and guide_path.exists():
        with open(guide_path, "r", encoding="utf-8") as f:
            guide_json = json.load(f)
        payload["conceptGuide"] = guide_json

    if not dry_run:
        slug = config["indicatorId"]
        es_data = {
            "question": config.get("question", "¿Cómo se componen los ingresos, el gasto público por función y el saldo fiscal en España?"),
            "unit": config.get("headlineUnit", "% del PIB"),
            "contextNotes": payload["contextNotes"],
            "sources": payload["sources"],
        }
        if "conceptGuide" in payload:
            es_data["conceptGuide"] = payload["conceptGuide"]

        en_data = {
            "question": "How are revenues, public spending by function, and fiscal balance composed in Spain?",
            "unit": "% of GDP",
            "contextNotes": [
                "Consolidated historical series of General Government Main Aggregates (S.13) according to ESA 2010.",
                "Fiscal balance (B.9) equals Total Revenue (TR) minus Total Expenditure (TE). A negative sign indicates a deficit and positive indicates a surplus.",
                "Functional expenditure breakdown follows international COFOG 1999 classification (GF01 to GF10). Public debt interest is accounted for under General Public Services (GF0107).",
                "Tax revenues stem from ESA 2010 aggregates (D.2, D.5, D.61, D.91). Non-tax revenues are computed by difference against TR."
            ],
            "sources": payload["sources"],
        }
        if "conceptGuide" in payload:
            en_data["conceptGuide"] = payload["conceptGuide"]

        publish_dataset(slug, payload, es_data, en_data)

    return payload

"""
Cliente HTTP y decodificador JSON-stat 2.0 para la API de Estadísticas de Eurostat v1.0.
Responsabilidad única (SRP): Transporte HTTP, reintentos con backoff exponencial, barra de progreso y decodificación genérica JSON-stat 2.0.
"""

from dataclasses import dataclass
import json
import math
import time
from typing import Any
import requests
from tqdm import tqdm


@dataclass(frozen=True)
class EurostatDataset:
    code: str
    updated: str | None          # Normalizado YYYY-MM-DD extraído de "updated" en JSON-stat
    records: list[dict[str, Any]] # Diccionarios con claves de dimensión + "value": float


def parse_jsonstat(payload: dict[str, Any], dataset_code: str) -> EurostatDataset:
    """
    Decodifica una respuesta JSON-stat 2.0 a una lista plana de registros numéricos.
    Implementa el algoritmo de decodificación exacto (soporta diccionarios dispersos o listas densas de valores,
    y índices de categorías en diccionario o lista).
    """
    dim_ids = payload.get("id")
    sizes = payload.get("size")
    dimensions = payload.get("dimension")
    raw_values = payload.get("value", {})

    if not dim_ids or not sizes or not dimensions or len(dim_ids) != len(sizes):
        raise ValueError(f"Estructura JSON-stat inválida o incompleta para el dataset {dataset_code}.")

    # 1. Construir lista ordenada de códigos de categoría por dimensión (pos_to_code)
    pos_to_code = []
    for dim_id in dim_ids:
        dim_obj = dimensions.get(dim_id, {})
        cat = dim_obj.get("category", {})
        cat_index = cat.get("index")

        if isinstance(cat_index, dict):
            # Invertir diccionario {code: position} ordenado por posición
            sorted_items = sorted(cat_index.items(), key=lambda kv: kv[1])
            codes = [code for code, _ in sorted_items]
        elif isinstance(cat_index, list):
            codes = cat_index
        else:
            # Fallback a las claves directas del objeto 'category' si no hay 'index'
            cat_ids = cat.get("label", {})
            codes = list(cat_ids.keys()) if isinstance(cat_ids, dict) else []

        if not codes:
            raise ValueError(f"No se pudieron extraer índices de categoría para la dimensión '{dim_id}' en {dataset_code}.")
        pos_to_code.append(codes)

    # 2. Calcular zancadas (strides) en orden row-major
    n_dims = len(sizes)
    strides = [1] * n_dims
    for i in range(n_dims - 2, -1, -1):
        strides[i] = strides[i + 1] * sizes[i + 1]

    # 3. Normalizar valores crudos a iterable de (flat_index, numeric_value)
    items_iter = []
    if isinstance(raw_values, dict):
        for k, v in raw_values.items():
            if v is not None:
                try:
                    fval = float(v)
                    if math.isfinite(fval):
                        items_iter.append((int(k), fval))
                except (ValueError, TypeError):
                    continue
    elif isinstance(raw_values, list):
        for idx, v in enumerate(raw_values):
            if v is not None:
                try:
                    fval = float(v)
                    if math.isfinite(fval):
                        items_iter.append((idx, fval))
                except (ValueError, TypeError):
                    continue

    # 4. Decodificar índices planos a coordenadas multidimensionales
    records = []
    for flat_idx, val in items_iter:
        record = {}
        for i in range(n_dims):
            size_i = sizes[i]
            stride_i = strides[i]
            code_idx = (flat_idx // stride_i) % size_i
            dim_code = dim_ids[i]
            record[dim_code] = pos_to_code[i][code_idx]
        record["value"] = val
        records.append(record)

    # 5. Extraer y normalizar fecha de actualización (updated)
    raw_updated = payload.get("updated")
    updated_str = None
    if isinstance(raw_updated, str) and len(raw_updated) >= 10:
        sub = raw_updated[:10]
        if sub[4] == "-" and sub[7] == "-":
            updated_str = sub

    return EurostatDataset(code=dataset_code, updated=updated_str, records=records)


def fetch_eurostat_dataset(
    base_url: str,
    dataset_code: str,
    params: dict[str, Any],
    since_year: int | None = None,
    timeout: int = 60,
    max_retries: int = 3,
) -> EurostatDataset:
    """
    Realiza una petición HTTP GET streaming a la API de Eurostat con barra de progreso tqdm
    y reintentos exponenciales ante códigos 429, 500, 502, 503, 504 o timeouts.
    """
    url = f"{base_url}/{dataset_code}"
    query_params = {
        "format": "JSON",
        "lang": "EN",
    }
    if since_year is not None:
        query_params["sinceTimePeriod"] = str(since_year)

    for k, v in params.items():
        query_params[k] = v

    print(f"\n🌐 Consultando Eurostat API v1.0 ({dataset_code})...")

    backoff = 2
    response = None
    for attempt in range(1, max_retries + 1):
        try:
            response = requests.get(url, params=query_params, stream=True, timeout=timeout)
            if response.status_code in (429, 500, 502, 503, 504):
                if attempt < max_retries:
                    print(f"      ⚠️ Servidor Eurostat devolvió HTTP {response.status_code}. Reintentando en {backoff}s (intento {attempt}/{max_retries})...")
                    time.sleep(backoff)
                    backoff *= 2
                    continue
            response.raise_for_status()
            break
        except (requests.exceptions.RequestException, TimeoutError) as e:
            if attempt < max_retries:
                print(f"      ⚠️ Error de red o timeout ({e}). Reintentando en {backoff}s (intento {attempt}/{max_retries})...")
                time.sleep(backoff)
                backoff *= 2
                continue
            raise RuntimeError(f"Fallo conectando con la API de Eurostat para {dataset_code} tras {max_retries} intentos: {e}") from e

    if response is None:
        raise RuntimeError(f"No se pudo obtener respuesta para {dataset_code}.")

    total_size = int(response.headers.get("content-length", 0))
    chunk_size = 16384
    chunks = []

    with tqdm(
        total=total_size if total_size > 0 else None,
        unit="B",
        unit_scale=True,
        unit_divisor=1024,
        desc=f"      Descarga {dataset_code}",
        ncols=80,
    ) as pbar:
        for chunk in response.iter_content(chunk_size=chunk_size):
            if chunk:
                chunks.append(chunk)
                pbar.update(len(chunk))

    raw_bytes = b"".join(chunks)
    print("      ⌛ Deserializando JSON-stat 2.0...")
    try:
        payload = json.loads(raw_bytes.decode("utf-8"))
    except json.JSONDecodeError as e:
        raise RuntimeError(
            f"Error deserializando JSON-stat de Eurostat ({dataset_code}). "
            f"Respuesta inválida o truncada ({len(raw_bytes)} bytes). Detalle: {e}"
        ) from e

    dataset = parse_jsonstat(payload, dataset_code)

def parse_legacy_ine(raw_data: list) -> dict:
    data = {}
    for serie in raw_data:
        nombre = serie.get("Nombre", "")
        if "Flujo de inmigraciones" in nombre and "Total Nacional" in nombre and not "Español" in nombre and not "Extranjero" in nombre:
            parts = [p.strip() for p in nombre.split(".")]
            c = parts[-1] if parts[-1] else parts[-2]
            if c:
                if c not in data:
                    data[c] = {}
                for d in serie.get("Data", []):
                    if d.get("Valor") is not None:
                        y = str(d["Anyo"])
                        if int(y) <= 2020:
                            data[c][y] = int(round(float(d["Valor"])))
    return data


def parse_modern_ine(raw_data: list) -> dict:
    data = {}
    for serie in raw_data:
        meta = serie.get("MetaData", [])
        es_ambos = any(m.get("Nombre") == "Ambos sexos" for m in meta)
        es_nac_tot = any(m.get("T3_Variable") == "Nacionalidad" and m.get("Nombre") == "Total" for m in meta)
        if es_ambos and es_nac_tot:
            geo_name = None
            for m in meta:
                if m.get("T3_Variable") in ("Total Nacional", "Países y Continentes", "País de procedencia", "País de nacimiento"):
                    geo_name = m.get("Nombre", "").strip()
                    break
            if geo_name:
                if geo_name not in data:
                    data[geo_name] = {}
                for d in serie.get("Data", []):
                    if d.get("Valor") is not None:
                        y = str(d["Anyo"])
                        if int(y) >= 2021:
                            data[geo_name][y] = int(round(float(d["Valor"])))
    return data

def validate_concept_guide(guide: dict) -> None:
    if not guide.get("title") or not isinstance(guide["title"], str):
        raise ValueError("conceptGuide.title must be a non-empty string.")
    categories = {c["id"] for c in guide.get("categories", [])}
    if not categories:
        raise ValueError("conceptGuide.categories must not be empty.")

    terms = guide.get("terms", [])
    if not terms or not isinstance(terms, list):
        raise ValueError("conceptGuide.terms must be a non-empty list.")

    seen_codes = set()
    for term in terms:
        code = term.get("code")
        if not code or code in seen_codes:
            raise ValueError(f"Código de concepto duplicado o inválido en guía: {code}")
        seen_codes.add(code)
        cat_id = term.get("categoryId")
        if cat_id not in categories:
            raise ValueError(f"Categoría inválida '{cat_id}' en término '{code}'")

    for term in terms:
        for rel in term.get("relatedCodes", []):
            if rel not in seen_codes:
                raise ValueError(f"relatedCode '{rel}' en término '{term.get('code')}' no existe en terms.")

                    act_val = act_lk.get((y, ecode))
                    empr_val = empr_lk.get((y, ecode))
                    unemp_val = unemp_lk.get((y, ecode))

                    share_emp = None
                    if emp_val is not None and total_emp is not None and total_emp > 0:
                        share_emp = round((emp_val / total_emp) * 100.0, 2)

                    if any(v is not None for v in (emp_val, act_val, empr_val, unemp_val)):
                        labor_block[gkey] = {
                            "employedThousands": round(emp_val, 1) if emp_val is not None else None,
                            "shareOfEmployedPct": share_emp,
                            "activityRatePct": round(act_val, 1) if act_val is not None else None,
                            "employmentRatePct": round(empr_val, 1) if empr_val is not None else None,
                            "unemploymentRatePct": round(unemp_val, 1) if unemp_val is not None else None,
                        }

            def_slices[def_key] = {
                "demographics": demo_block,
                "laborMarket": labor_block,
            }

        has_any_income = any(
            (y, code, m) in inc_lookup
            for code in ("TOTAL", "NAT", "FOR")
            for m in ("MED", "MEAN")
        ) or any((y, code) in pov_lookup for code in ("TOTAL", "NAT", "FOR"))

        if not has_any_income:
            living_block = None
        else:
            living_block = {}
            target_income_groups = [
                ("total", "TOTAL"),
                ("nationals", "NAT"),
                ("foreigners", "FOR"),
                ("euForeigners", "EU27_2020_FOR"),
                ("nonEuForeigners", "NEU27_2020_FOR"),
            ]
            for gkey, ecode in target_income_groups:
                med_inc = inc_lookup.get((y, ecode, "MED"))
                mean_inc = inc_lookup.get((y, ecode, "MEAN"))
                pov_pct = pov_lookup.get((y, ecode))

                if any(v is not None for v in (med_inc, mean_inc, pov_pct)):
                    living_block[gkey] = {
                        "medianEquivalisedIncomeEur": round(med_inc, 1) if med_inc is not None else None,
                        "meanEquivalisedIncomeEur": round(mean_inc, 1) if mean_inc is not None else None,
                        "atRiskOfPovertyPct": round(pov_pct, 1) if pov_pct is not None else None,
                    }

        nat_val = acq_lookup.get(y)

        annual_records.append({
            "year": y,
            "naturalizations": nat_val,
            "definitions": def_slices,
            "livingConditions": living_block,

        cit_demo = def_slices["citizenship"]["demographics"]
        if cit_demo and "foreigners" in cit_demo:

"""
Cliente HTTP dedicado a consultar las tablas de la API oficial del INE.
Responsabilidad única (SRP): Comunicación e integración con la API Tempus JS del INE.
"""

import json
import requests
from tqdm import tqdm

INE_API_BASE_URL = "https://servicios.ine.es/wstempus/js/ES/DATOS_TABLA"


def fetch_ine_table(table_id: str, tip: str = "AM", timeout: int = 60) -> list:
    """
    Consulta la API Tempus del INE para obtener los datos crudos de una tabla determinada,
    mostrando una barra de progreso en tiempo real con tqdm.
    """
    url = f"{INE_API_BASE_URL}/{table_id}?tip={tip}"
    print(f"\n[1/4] 🌐 Descargando datos desde la API del INE (Tabla {table_id})...")

    with requests.get(url, stream=True, timeout=timeout) as response:
        response.raise_for_status()
        total_size = int(response.headers.get("content-length", 0))
        chunk_size = 16384
        chunks = []

        with tqdm(
            total=total_size if total_size > 0 else None,
            unit="B",
            unit_scale=True,
            unit_divisor=1024,
            desc="      Descarga INE",
            ncols=80,
        ) as pbar:
            for chunk in response.iter_content(chunk_size=chunk_size):
                if chunk:
                    chunks.append(chunk)
                    pbar.update(len(chunk))

        raw_bytes = b"".join(chunks)
        print("      ⌛ Deserializando respuesta JSON...")
        try:
            data = json.loads(raw_bytes.decode("utf-8"))
        except json.JSONDecodeError as e:
            raise RuntimeError(
                f"Error deserializando JSON de la tabla INE {table_id}. "
                f"La respuesta fue truncada o es inválida ({len(raw_bytes)} bytes recibidos). "
                f"Detalle: {e}"
            ) from e
        print(f"      ✓ Respuesta recibida: {len(data)} series de datos crudas obtenidas.")
        return data

def fetch(
    url: str,
    *,
    method: str = "GET",
    json_body: dict[str, Any] | None = None,
    allow_404: bool = False,
) -> requests.Response | None:
    last_error: Exception | None = None
    for attempt in range(RETRIES):
        try:
            if method == "POST":
                response = requests.post(url, json=json_body, headers=HEADERS, timeout=TIMEOUT)
            else:
                response = requests.get(url, headers=HEADERS, timeout=TIMEOUT)

            if allow_404 and response.status_code == 404:
                return None

            if response.status_code in {429, 500, 502, 503, 504} and attempt < RETRIES - 1:
                time.sleep(2 ** attempt)
                continue

            response.raise_for_status()
            return response
        except requests.RequestException as exc:
            last_error = exc
            if attempt < RETRIES - 1:
                time.sleep(2 ** attempt)
            else:
                raise
    raise RuntimeError(f"Error conectando con {url}: {last_error}")


def normalized_label(value: str) -> str:
    return re.sub(r"\s+", " ", value.replace("\xa0", " ")).strip()


def parse_number(value: str) -> int | float | None:
    raw = str(value).strip().replace("\xa0", " ")
    if raw in {"", "-", "—", "…", "..", ":", "n.d.", "nd"}:
        return None
    raw = raw.replace(" ", "")
    if "," in raw and "." in raw:
        if raw.rfind(",") > raw.rfind("."):
            raw = raw.replace(".", "").replace(",", ".")
        else:
            raw = raw.replace(",", "")
    elif "," in raw:
        raw = raw.replace(".", "").replace(",", ".")
    else:
        if re.fullmatch(r"-?\d{1,3}(\.\d{3})+", raw):
            raw = raw.replace(".", "")
    try:
        number = float(raw)
    except ValueError:
        return None
    if not math.isfinite(number):
        return None
    return int(number) if number.is_integer() else round(number, 2)


def decode_text_bytes(content: bytes) -> str:
    for encoding in ("utf-8-sig", "cp1252", "latin-1"):
        try:
            return content.decode(encoding)
        except UnicodeDecodeError:
            continue
    raise ValueError("No se pudo decodificar la respuesta textual de la fuente.")


def discover_aeat_edition_urls(start_year: int = START_YEAR) -> dict[str, list[tuple[str, str]]]:
    """
    Sondea las ediciones anuales de la AEAT desde `start_year` hasta el año actual
    y extrae pares (url, link_label) para las tablas nacionales por edad y sexo
    (Asalariados, Desempleados y Pensionistas).
    """
    current_year = date.today().year
    discovered: dict[str, list[tuple[str, str]]] = {}
    exclude_words = [
        "comunidad",
        "provincia",
        "sector",
        "tramo",
        "pagador",
        "módulo",
        "modulo",
        "entidades",
        "ccaa",
        "centila",
        "decila",
    ]

    for year in range(start_year, current_year + 1):
        map_url = AEAT_MAP_URL_TEMPLATE.format(year=year)
        resp = fetch(map_url, allow_404=True)
        if resp is None:
            continue

        soup = BeautifulSoup(decode_text_bytes(resp.content), "html.parser")
        year_links: list[tuple[str, str]] = []
        seen_urls: set[str] = set()

        for anchor in soup.find_all("a", href=True):
            if anchor["href"].startswith("javascript"):
                continue
            label = normalized_label(anchor.get_text(" ", strip=True))
            label_cf = label.casefold()

            if any(w in label_cf for w in exclude_words):
                continue

            if (
                any(pop in label_cf for pop in ("asalariados", "desempleados", "pensionistas"))
                and "edad" in label_cf
                and "sexo" in label_cf
            ):
                candidate = urljoin(map_url, anchor["href"])
                if candidate not in seen_urls:
                    seen_urls.add(candidate)
                    year_links.append((candidate, label))

        if year_links:
            discovered[str(year)] = year_links

    return discovered


def _match_population_spec(
    soup: BeautifulSoup, link_label: str, table_header_text: str
) -> dict[str, Any] | None:
    """
    Identifica la población (asalariados, desempleados o pensionistas) exclusivamente
    a partir del enlace origen, el título/encabezados propios de la tabla o su cabecera,
    evitando falsos positivos del menú de navegación global de la AEAT.
    """
    title_text = normalized_label(soup.title.get_text(" ", strip=True)) if soup.title else ""
    headings_text = " ".join(
        normalized_label(h.get_text(" ", strip=True))
        for h in soup.find_all(["h1", "h2", "h3", "h4", "caption"])
    )
    candidates = [link_label, title_text, headings_text, table_header_text]
    for candidate in candidates:
        cf = candidate.casefold()
        if "asalariad" in cf:
            return POPULATION_SPECS["asalariados"]
        if "desemplead" in cf or "desempleo" in cf:
            return POPULATION_SPECS["desempleados"]
        if "pensionist" in cf or "pensiones" in cf:
            return POPULATION_SPECS["pensionistas"]
    return None


def _detect_sex_from_page(soup: BeautifulSoup, page_text: str, link_label: str) -> str | None:
    m_link = re.search(
        r"Sexo\s*:?\s*(Total|Ambos sexos|Var[oó]n|Hombres?|Mujeres?|No consta)\b",
        link_label,
        re.I,
    )
    if m_link:
        return SEX_ALIASES.get(normalized_label(m_link.group(1)).casefold())

    sex_match = re.search(
        r"Sexo\s*:?\s*(Total|Ambos sexos|Var[oó]n|Hombres?|Mujeres?|No consta)\b",
        page_text,
        re.I,
    )
    if sex_match:
        return SEX_ALIASES.get(normalized_label(sex_match.group(1)).casefold())

    for select in soup.find_all("select"):
        sel_meta = f"{select.get('id', '')} {select.get('name', '')} {select.get('title', '')}".casefold()
        opt = select.find("option", selected=True)
        if opt:
            opt_txt = normalized_label(opt.get_text(" ", strip=True)).casefold()
            if "sexo" in sel_meta or opt_txt in SEX_ALIASES:
                return SEX_ALIASES.get(opt_txt)

    return None

        opt = select.find("option", selected=True)
        if opt:
            opt_txt = normalized_label(opt.get_text(" ", strip=True)).casefold()
            if "nac" in sel_meta or opt_txt in NATIONALITY_ALIASES:
                return NATIONALITY_ALIASES.get(opt_txt)

    heading = soup.find(lambda tag: tag.name in ["h2", "h3", "h4", "div", "p"] and "nacionalidad" in tag.get_text().lower())
    if heading:
        txt = normalized_label(heading.get_text(" ", strip=True)).casefold()
        m = re.search(r"nacionalidad\s*:?\s*(total|española|espanola|extranjera)", txt)
        if m:
            return NATIONALITY_ALIASES.get(m.group(1))

    m_link = re.search(
        r"Nacionalidad\s*:?\s*(Total|Española|Extranjera|Unión Europea|Resto de Europa)\b",
        link_label,
        re.I,
    )
    if m_link:
        return NATIONALITY_ALIASES.get(normalized_label(m_link.group(1)).casefold())

    nat_match = re.search(
        r"Nacionalidad\s*:?\s*(Total|Española|Extranjera|Unión Europea|Resto de Europa)\b",
        page_text,
        re.I,
    )
    if nat_match:
        return NATIONALITY_ALIASES.get(normalized_label(nat_match.group(1)).casefold())

    return None


def parse_aeat_page(
    url: str, html: str, link_label: str = ""
) -> tuple[list[dict[str, Any]], list[tuple[str, str]]]:
    """
    Analiza una página de tabla de la AEAT y devuelve:
      - Lista de bloques extraídos (por sexo para la nacionalidad de la página).
      - Lista de enlaces hermanos (url, label) si la página contiene pestañas/enlaces
        a otras nacionalidades (p. ej. Española / Extranjera).
    """
    soup = BeautifulSoup(html, "html.parser")
    page_text = normalized_label(soup.get_text(" ", strip=True))

    sibling_links: list[tuple[str, str]] = []
    for a in soup.find_all("a", href=True):
        a_text = normalized_label(a.get_text(" ", strip=True))
        a_cf = a_text.casefold()
        if a_cf in {"española", "extranjera", "nacionalidad española", "nacionalidad extranjera"}:
            cand = urljoin(url, a["href"])
            if cand != url and not a["href"].startswith("javascript"):
                sibling_links.append((cand, f"{link_label} Nacionalidad: {a_text}"))

    year_match = re.search(r"Ejercicio\s+(20\d{2})", page_text, re.I) or re.search(r"/(20\d{2})/", url)
    if not year_match:
        return [], sibling_links
    year = year_match.group(1)

    page_nat_key: str | None = None
    lbl_cf = link_label.casefold()
    if "nacionalidad: española" in lbl_cf or "nacionalidad: espanola" in lbl_cf:
        page_nat_key = "spanish"
    elif "nacionalidad: extranjera" in lbl_cf:
        page_nat_key = "foreign"
    elif "nacionalidad: total" in lbl_cf:
        page_nat_key = "total"

    if not page_nat_key:
        heading = soup.find(
            lambda tag: tag.name in ["h2", "h3", "h4", "div", "p"]
            and "nacionalidad" in tag.get_text().lower()
        )
        if heading:
            txt = normalized_label(heading.get_text(" ", strip=True)).casefold()
            m = re.search(r"nacionalidad\s*:?\s*(total|española|espanola|extranjera)", txt)
            if m:
                page_nat_key = NATIONALITY_ALIASES.get(m.group(1))

    page_nat_key = page_nat_key or "total"

    extracted_blocks: list[dict[str, Any]] = []

    for table in soup.find_all("table"):
        parsed_rows: list[list[str]] = []
        for tr in table.find_all("tr"):
            cells = [normalized_label(c.get_text(" ", strip=True)) for c in tr.find_all(["th", "td"])]
            if cells:
                parsed_rows.append(cells)

        if len(parsed_rows) < 2:
            continue

        table_header_text = " ".join(" ".join(r) for r in parsed_rows[:5])
        matched_spec = _match_population_spec(soup, link_label, table_header_text)
        if not matched_spec:
            continue

        measure_names = matched_spec["measures"]
        n_meas = len(measure_names)
        pop_k = matched_spec["key"]

        sex_sections: dict[str, dict[str, dict[str, int | float]]] = {
            "total": {},
            "male": {},
            "female": {},
        }

        for row in parsed_rows:
            first_cell_cf = row[0].casefold()
            numeric_tail = [parse_number(v) for v in row[1:]]
            if all(v is None for v in numeric_tail):
                continue

            age_label = (
                "Total"
                if first_cell_cf == "esquina" or first_cell_cf.startswith("total")
                else row[0]
            )

            # Caso A: Fila con 1 columna de dimensión + 3 bloques de métricas por sexo (Total, Varón, Mujer)
            if len(row) >= 1 + 3 * n_meas:
                vals = [parse_number(v) for v in row[1 : 1 + 3 * n_meas]]
                if all(v is not None for v in vals):
                    for idx_sex, sex_code in enumerate(("total", "male", "female")):
                        chunk = vals[idx_sex * n_meas : (idx_sex + 1) * n_meas]
                        row_dict = {
                            m_name: chunk[i]  # type: ignore[assignment]
                            for i, m_name in enumerate(measure_names)
                        }
                        if pop_k == "unemploymentRecipients":
                            row_dict["totalBenefitsEur"] = round(row_dict["recipients"] * row_dict["averageBenefitEur"])
                        elif pop_k == "pensionRecipients":
                            row_dict["totalPensionsEur"] = round(row_dict["recipients"] * row_dict["averagePensionEur"])
                        sex_sections[sex_code][age_label] = row_dict
                    continue

            # Caso B: Fila estándar de 1 columna de etiqueta + 1 bloque de métricas (sexo total)
            if len(row) >= 1 + n_meas:
                vals = [parse_number(v) for v in row[1 : 1 + n_meas]]
                if all(v is not None for v in vals):
                    row_dict = {
                        m_name: vals[i]  # type: ignore[assignment]
                        for i, m_name in enumerate(measure_names)
                    }
                    if pop_k == "unemploymentRecipients":
                        row_dict["totalBenefitsEur"] = round(row_dict["recipients"] * row_dict["averageBenefitEur"])
                    elif pop_k == "pensionRecipients":
                        row_dict["totalPensionsEur"] = round(row_dict["recipients"] * row_dict["averagePensionEur"])
                    sex_sections["total"][age_label] = row_dict

        for sex_code, age_bands in sex_sections.items():
            if "Total" in age_bands:
                extracted_blocks.append({
                    "year": year,
                    "populationKey": pop_k,
                    "nationalityKey": page_nat_key,
                    "sexKey": sex_code,
                    "ageBands": age_bands,
                    "url": url,
                })

        if extracted_blocks:
            break

    return extracted_blocks, sibling_links


def fetch_all_aeat_data() -> tuple[dict[str, dict[str, Any]], list[str]]:
    """
    Descarga y consolida todas las ediciones anuales de la AEAT que comparten
    una estructura común de poblaciones × nacionalidad × sexo.
    """
    print("\n[1/3] 🌐 Descubriendo y descargando ediciones anuales de la AEAT...")
    edition_links = discover_aeat_edition_urls(START_YEAR)
    if not edition_links:
        raise RuntimeError("No se encontraron ediciones anuales en el portal de la AEAT.")

    raw_by_year: dict[str, dict[str, Any]] = {}
    urls_by_year: dict[str, list[str]] = {}

    for year in sorted(edition_links.keys(), key=int):
        year_bucket: dict[str, Any] = {}
        year_urls_used: list[str] = []
        queue = list(edition_links[year])
        visited: set[str] = set()

        while queue:
            url, link_label = queue.pop(0)
            if url in visited:
                continue
            visited.add(url)

            resp = fetch(url, allow_404=True)
            if resp is None:
                continue

            blocks, siblings = parse_aeat_page(url, decode_text_bytes(resp.content), link_label=link_label)
            for sib_url, sib_label in siblings:
                if sib_url not in visited:
                    queue.append((sib_url, sib_label))

            for parsed in blocks:
                pop_k = parsed["populationKey"]
                nat_k = parsed["nationalityKey"]
                sex_k = parsed["sexKey"]
                existing = year_bucket.setdefault(pop_k, {}).setdefault(nat_k, {}).get(sex_k, {})
                # Conservar la versión más detallada (con tramos de edad si existe)
                if len(parsed["ageBands"]) >= len(existing):
                    year_bucket[pop_k][nat_k][sex_k] = parsed["ageBands"]
                if url not in year_urls_used:
                    year_urls_used.append(url)

        # Si pensionRecipients solo tiene 'total', asegurar que se popule también para 'spanish' y 'foreign'
        if "pensionRecipients" in year_bucket:
            pen_map = year_bucket["pensionRecipients"]
            if "total" in pen_map:
                if "spanish" not in pen_map:
                    pen_map["spanish"] = pen_map["total"]
                if "foreign" not in pen_map:
                    pen_map["foreign"] = pen_map["total"]

        if year_bucket:
            raw_by_year[year] = year_bucket
            urls_by_year[year] = year_urls_used
            pops_found = ", ".join(sorted(year_bucket.keys()))
            nats_found = sorted({n for p in year_bucket.values() for n in p.keys()})
            print(f"      ✓ AEAT {year}: {len(year_urls_used)} páginas ({pops_found} | nacionalidades: {nats_found}).")
        else:
            print(f"      ⚠️ AEAT {year}: ninguna tabla coincidió con el esquema esperado.")

    if not raw_by_year:
        raise RuntimeError("Ningún año de la AEAT pudo ser extraído.")

    valid_pops_per_year: dict[str, set[str]] = {}
    for year, bucket in raw_by_year.items():
        valid_pops = {
            pop_k
            for pop_k, nat_map in bucket.items()
            if "spanish" in nat_map
            and "foreign" in nat_map
            and "total" in nat_map["spanish"]
            and "total" in nat_map["foreign"]
        }
        if valid_pops:
            valid_pops_per_year[year] = valid_pops

    if not valid_pops_per_year:
        raise RuntimeError("Ningún año de la AEAT contiene desglose completo por nacionalidad española y extranjera.")

    common_populations = set.intersection(*valid_pops_per_year.values())
    if not common_populations:
        all_pops = {p for pops in valid_pops_per_year.values() for p in pops}
        common_populations = {
            p for p in all_pops
            if sum(1 for pops in valid_pops_per_year.values() if p in pops) >= max(2, len(valid_pops_per_year) // 2)
        }

    by_year: dict[str, dict[str, Any]] = {}
    used_urls: list[str] = []
    for year in sorted(valid_pops_per_year.keys(), key=int):
        if common_populations.issubset(valid_pops_per_year[year]):
            by_year[year] = {
                pop_k: raw_by_year[year][pop_k]
                for pop_k in sorted(common_populations)
            }
            used_urls.extend(urls_by_year[year])
        else:
            print(f"      ⚠️ AEAT {year}: omitido por no incluir todas las poblaciones comunes ({sorted(common_populations)}).")

    if not by_year:
        raise RuntimeError("Ningún año de la AEAT cumplió con la estructura común esperada.")

    return by_year, used_urls


def parse_tgss_csv_text(text: str) -> dict[str, dict[str, Any]]:
    """
    Decodifica el CSV de PX-Web de la TGSS y agrupa los datos por año natural,
    conservando todos los períodos mensuales/anuales disponibles y el cierre de cada año.
    """
    rows = list(csv.reader(io.StringIO(text), delimiter=";", quotechar='"'))
    if len(rows) < 3:
        raise ValueError(f"CSV de la TGSS demasiado corto ({len(rows)} filas).")

    header_index = None
    for idx, row in enumerate(rows[:10]):
        labels = [normalized_label(c).casefold() for c in row]
        if len(labels) >= 3 and labels[0] in {"período", "periodo"} and labels[1] in {"país", "pais"}:
            header_index = idx
            break

    if header_index is None:
        raise ValueError("No se encontró la fila de cabecera (Período;País;...) en el CSV de la TGSS.")

    headers = [normalized_label(c) for c in rows[header_index]]
    regime_columns = headers[2:]
    if not any("regimen" in h.casefold() or "régimen" in h.casefold() for h in regime_columns):
        raise ValueError(f"Columnas de régimen inesperadas en TGSS: {regime_columns}")

    periods_by_year: dict[str, dict[str, dict[str, dict[str, int | float]]]] = {}

    for line_num, row in enumerate(rows[header_index + 1 :], start=header_index + 2):
        if not row or not any(c.strip() for c in row):
            continue
        if len(row) != len(headers):
            raise ValueError(f"Fila {line_num} de TGSS tiene {len(row)} columnas (esperadas {len(headers)}).")

        raw_period = normalized_label(row[0])
        raw_country = normalized_label(row[1])
        if not raw_period or not raw_country:
            continue

        if re.fullmatch(r"\d{8}", raw_period):
            period_iso = datetime.strptime(raw_period, "%Y%m%d").date().isoformat()
            year = period_iso[:4]
        elif re.fullmatch(r"20\d{2}", raw_period):
            period_iso = raw_period
            year = raw_period
        else:
            continue

        if int(year) < START_YEAR:
            continue

        country_dict: dict[str, int | float] = {}
        for regime_label, raw_val in zip(regime_columns, row[2:]):
            val = parse_number(raw_val)
            if val is not None:
                clean_regime = re.sub(r"\s+TOTAL\s+SEXO$", "", regime_label, flags=re.I).strip()
                country_dict[clean_regime] = val

        if country_dict:
            periods_by_year.setdefault(year, {}).setdefault(period_iso, {})[raw_country] = country_dict

    if not periods_by_year:
        raise ValueError("El CSV de la TGSS no produjo registros numéricos válidos.")

    result_by_year: dict[str, dict[str, Any]] = {}
    for year, p_map in sorted(periods_by_year.items(), key=lambda kv: int(kv[0])):
        latest_period = max(p_map.keys())
        result_by_year[year] = {
            "referencePeriod": latest_period,
            "periodsAvailable": sorted(p_map.keys()),
            "affiliatesByCountryAndRegime": p_map[latest_period],
        }

    return result_by_year


def fetch_all_tgss_data() -> tuple[dict[str, dict[str, Any]], str]:
    """
    Consulta la API PX-Web de la TGSS solicitando todos los períodos y países disponibles
    en formato CSV semicolon-delimited, con respaldo en la consulta guardada oficial.
    """
    print("\n[2/3] 🌐 Descargando serie histórica de afiliados extranjeros de la TGSS...")

    try:
        meta_resp = fetch(TGSS_PXWEB_TABLE_URL)
        if meta_resp is not None:
            meta = meta_resp.json()
            variables = meta.get("variables", [])
            if variables:
                query_items = []
                for var in variables:
                    code = var["code"]
                    values = var.get("values", [])
                    if "sexo" in code.casefold():
                        total_vals = [
                            v for v, t in zip(values, var.get("valueTexts", []))
                            if "total" in t.casefold()
                        ]
                        selected = total_vals if total_vals else values
                    else:
                        selected = values
                    query_items.append({
                        "code": code,
                        "selection": {"filter": "item", "values": selected},
                    })

                px_payload = {
                    "query": query_items,
                    "response": {"format": "csv"},
                }
                csv_resp = fetch(TGSS_PXWEB_TABLE_URL, method="POST", json_body=px_payload)
                if csv_resp is not None:
                    parsed = parse_tgss_csv_text(decode_text_bytes(csv_resp.content))
                    print(f"      ✓ TGSS PX-Web API decodificada: {len(parsed)} años obtenidos.")
                    return parsed, TGSS_PORTAL_URL
    except Exception as exc:
        print(f"      ⚠️ Consulta PX-Web API directa no disponible ({exc}); usando enlace oficial de consulta...")

    fallback_resp = fetch(TGSS_FALLBACK_CSV_URL)
    if fallback_resp is None:
        raise RuntimeError("No se pudo descargar el CSV oficial de la TGSS.")
    parsed = parse_tgss_csv_text(decode_text_bytes(fallback_resp.content))
    print(f"      ✓ TGSS CSV oficial decodificado: {len(parsed)} años obtenidos.")
    return parsed, TGSS_FALLBACK_CSV_URL


def build_payload(
    aeat_by_year: dict[str, dict[str, Any]],
    tgss_by_year: dict[str, dict[str, Any]],
    tgss_source_url: str,
) -> dict[str, Any]:
    """
    Construye el JSON canónico unificado y valida invariantes numéricas y estructurales.
    """
    print("\n[3/3] ⚖️ Consolidando y validando registros administrativos...")
    all_years = sorted(set(aeat_by_year.keys()) | set(tgss_by_year.keys()), key=int)
    if not all_years:
        raise ValueError("No hay años disponibles para consolidar.")

    annual_records: list[dict[str, Any]] = []
    data_points: list[dict[str, Any]] = []

    for year in all_years:
        aeat_rec = aeat_by_year.get(year)
        tgss_rec = tgss_by_year.get(year)

        if aeat_rec:
            for pop_key, nat_map in aeat_rec.items():
                for nat_key, sex_map in nat_map.items():
                    for sex_key, age_map in sex_map.items():
                        for age_label, metrics in age_map.items():
                            for m_name, m_val in metrics.items():
                                if not isinstance(m_val, (int, float)) or not math.isfinite(m_val) or m_val < 0:
                                    raise ValueError(
                                        f"Valor corrupto o negativo en AEAT {year} {pop_key}/{nat_key}/{sex_key}/{age_label}: {m_name}={m_val}"
                                    )

        headline_foreign_affiliates: int | float | None = None
        if tgss_rec:
            aff_map = tgss_rec["affiliatesByCountryAndRegime"]
            for country, regimes in aff_map.items():
                for regime, count in regimes.items():
                    if not isinstance(count, (int, float)) or not math.isfinite(count) or count < 0:
                        raise ValueError(
                            f"Valor corrupto o negativo en TGSS {year} {country}/{regime}: {count}"
                        )

            total_country = aff_map.get("TOTAL") or aff_map.get("Total")
            if total_country:
                for reg_name, val in total_country.items():
                    if "total" in reg_name.casefold():
                        headline_foreign_affiliates = val
                        break
                if headline_foreign_affiliates is None:
                    headline_foreign_affiliates = next(iter(total_country.values()), None)

        annual_records.append({
            "year": year,
            "aeat": aeat_rec,
            "tgss": tgss_rec,
        })

        if headline_foreign_affiliates is not None:
            data_points.append({
                "year": year,
                "value": headline_foreign_affiliates,
            })

    if not data_points:
        raise ValueError("No se pudieron construir dataPoints cronológicos principales.")

    today_str = date.today().isoformat()
    aeat_years = sorted(aeat_by_year.keys(), key=int)
    tgss_years = sorted(tgss_by_year.keys(), key=int)

    sources = [
        {
            "institution": "Agencia Estatal de Administración Tributaria (AEAT)",
            "shortName": "AEAT",
            "title": "Mercado de trabajo y pensiones en las fuentes tributarias (Nacionalidad y Sexo)",
            "period": f"{aeat_years[0]}–{aeat_years[-1]}",
            "lastUpdated": today_str,
            "url": AEAT_PORTAL_URL,
        },
        {
            "institution": "Tesorería General de la Seguridad Social (TGSS)",
            "shortName": "TGSS",
            "title": "Afiliados extranjeros a la Seguridad Social (último día del período por país y régimen)",
            "period": f"{tgss_years[0]}–{tgss_years[-1]}",
            "lastUpdated": today_str,
            "url": tgss_source_url,
        },
    ]

    context_notes_es = [
        "Datos procedentes directamente de registros administrativos oficiales: declaraciones tributarias (Modelo 190 de la AEAT) y ficheros de afiliación de la Tesorería General de la Seguridad Social (TGSS), sin imputaciones ni estimaciones muestrales.",
        "Las magnitudes de la AEAT (asalariados, perceptores de prestaciones por desempleo y pensionistas) incluyen retribuciones/prestaciones brutas anuales y medias por perceptor sin ajuste a jornada completa.",
        "Los datos de la TGSS corresponden al número de afiliados extranjeros en alta laboral el último día del período de referencia disponible para cada año.",
        "Se incluyen exclusivamente los ejercicios anuales que comparten una estructura homogénea de variables y desgloses en las publicaciones oficiales.",
    ]

    return {
        "id": SLUG,
        "kind": "admin-registers-nationality",
        "category": "economy-employment",
        "valueFormat": "number",
        "lastUpdated": today_str,
        "dataPoints": data_points,
        "annualRecords": annual_records,
        "contextNotes": context_notes_es,
        "sources": sources,
    }


def run_pipeline() -> dict[str, Any]:
    aeat_by_year, _ = fetch_all_aeat_data()
    tgss_by_year, tgss_url = fetch_all_tgss_data()
    payload = build_payload(aeat_by_year, tgss_by_year, tgss_url)

    es_data = {
        "question": "¿Cuántos trabajadores extranjeros afiliados, asalariados, desempleados y pensionistas registran la Seguridad Social y la Agencia Tributaria en España?",
        "unit": "afiliados extranjeros",
        "contextNotes": payload["contextNotes"],
        "sources": payload["sources"],
    }

    en_data = {
        "question": "How many foreign Social Security affiliates, salaried workers, unemployment recipients, and pensioners are recorded by Spain's tax and Social Security registers?",
        "unit": "foreign affiliates",
        "contextNotes": [
            "Data sourced directly from official Spanish administrative registers: tax withholding filings (AEAT Form 190) and Social Security affiliation records (TGSS), without survey sampling or synthetic imputation.",
            "AEAT figures (salaried workers, unemployment benefit recipients, and pensioners) report annual gross amounts and unadjusted per-recipient averages (not full-time equivalent).",
            "TGSS figures reflect foreign nationals actively registered with Social Security on the last day of the latest available reference period for each year.",
            "Only annual releases sharing a common variable and breakdown schema across official publications are included.",
        ],
        "sources": payload["sources"],
    }

    publish_dataset(SLUG, payload, es_data, en_data)
    return payload

"""
Módulo de utilidades para scrapers y ETL de Factores:
- Normalización de cadenas y conversión a slugs.
- Validación de integridad y persistencia en staging/producción.
- Publicación atómica de datasets divididos en datos numéricos y traducciones.
"""

from datetime import date
import json
import os
from pathlib import Path
import re
import shutil


def slugify(text: str) -> str:
    """Transforma un texto o nombre en un identificador limpio (slug) sin acentos ni caracteres especiales."""
    text = text.lower().strip()
    replacements = {
        "á": "a", "é": "e", "í": "i", "ó": "o", "ú": "u",
        "ñ": "n", "ü": "u", "(": "", ")": "", ",": ""
    }
    for k, v in replacements.items():
        text = text.replace(k, v)
    return re.sub(r"[^a-z0-9]+", "-", text).strip("-")


def validate_and_stage(
    items: dict | list[dict],
    staging_dir: Path = Path("./staging"),
) -> bool:
    """
    Verifica que las tarjetas y sus series no estén vacías antes de guardarlas en staging.
    """
    print(f"\n[3/4] 🧪 Validando tarjetas y guardando datos en ({staging_dir}/)...")
    items_staging_dir = staging_dir / "items"
    items_staging_dir.mkdir(parents=True, exist_ok=True)

    items_list = items if isinstance(items, list) else [items]
    existing_staging_entries = {}

    staging_index_path = staging_dir / "items.json"
    if staging_index_path.exists():
        try:
            with open(staging_index_path, "r", encoding="utf-8") as f:
                for entry in json.load(f):
                    existing_staging_entries[entry["id"]] = entry
        except Exception:
            pass

    for item_payload in items_list:
        item_id = item_payload.get("id")
        if not item_id:
            print("\n      ❌ ERROR: La tarjeta no tiene 'id'")
            return False

        with open(items_staging_dir / f"{item_id}.json", "w", encoding="utf-8") as f:
            json.dump(item_payload, f, ensure_ascii=False, indent=2)

        existing_staging_entries[item_id] = {
            "id": item_id,
            "category": item_payload.get("category", "economy-employment"),
            "question": item_payload.get("question", item_id),
            "unit": item_payload.get("unit", ""),
            "lastUpdated": item_payload.get("lastUpdated") or date.today().isoformat(),
        }

    with open(staging_dir / "items.json", "w", encoding="utf-8") as f:
        json.dump(list(existing_staging_entries.values()), f, ensure_ascii=False, indent=2)

    print(f"      ✓ Validación superada: {len(items_list)} tarjeta(s) guardada(s) en {staging_dir}/items/")
    return True


def publish_dataset(
    slug: str,
    dataset_data: dict,
    es_data: dict,
    en_data: dict,
    frontend_data_dir: Path | None = None,
) -> None:
    """
    Publica atómicamente un dataset dividido en:
    - Pure numerical data -> frontend/data/datasets/{slug}.json
    - Spanish copy -> frontend/data/locales/es/{slug}.json
    - English copy -> frontend/data/locales/en/{slug}.json
    """
    if frontend_data_dir is None:
        frontend_data_dir = Path(__file__).resolve().parent.parent / "frontend" / "data"

    datasets_dir = frontend_data_dir / "datasets"
    locales_es_dir = frontend_data_dir / "locales" / "es"
    locales_en_dir = frontend_data_dir / "locales" / "en"

    datasets_dir.mkdir(parents=True, exist_ok=True)
    locales_es_dir.mkdir(parents=True, exist_ok=True)
    locales_en_dir.mkdir(parents=True, exist_ok=True)

    def _atomic_write(target_path: Path, data: dict):
        tmp_path = target_path.with_suffix(".tmp")
        with open(tmp_path, "w", encoding="utf-8") as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        os.replace(tmp_path, target_path)

    _atomic_write(datasets_dir / f"{slug}.json", dataset_data)
    _atomic_write(locales_es_dir / f"{slug}.json", es_data)
    _atomic_write(locales_en_dir / f"{slug}.json", en_data)

    print(f"      ✓ Dataset '{slug}' publicado correctamente en {frontend_data_dir}")


def promote_to_data(
    prefix_filter: str = "",
    staging_dir: Path = Path("./staging"),
    data_dir: Path = Path("./data"),
) -> None:
    """
    Sincroniza las tarjetas en staging con la estructura de producción.
    """
    print(f"\n[4/4] 🚀 Promocionando datos a producción...")
    items_staging_dir = staging_dir / "items"
    if not items_staging_dir.exists():
        return

    for src_file in items_staging_dir.glob("*.json"):
        slug = src_file.stem
        if prefix_filter and not slug.startswith(prefix_filter):
            continue
        with open(src_file, "r", encoding="utf-8") as f:
            content = json.load(f)

        question = content.get("question", "")
        unit = content.get("unit", "")
        context_notes = content.get("contextNotes", [])
        sources = content.get("sources", [])
        concept_guide = content.get("conceptGuide")

        es_data = {
            "question": question.get("es", question) if isinstance(question, dict) else question,
            "unit": unit.get("es", unit) if isinstance(unit, dict) else unit,
            "contextNotes": context_notes.get("es", context_notes) if isinstance(context_notes, dict) else context_notes,
            "sources": sources,
        }
        if concept_guide:
            es_data["conceptGuide"] = concept_guide

        en_data = {
            "question": question.get("en", question) if isinstance(question, dict) else question,
            "unit": unit.get("en", unit) if isinstance(unit, dict) else unit,
            "contextNotes": context_notes.get("en", context_notes) if isinstance(context_notes, dict) else context_notes,
            "sources": sources,
        }
        if concept_guide:
            en_data["conceptGuide"] = concept_guide

        dataset_data = dict(content)

        publish_dataset(slug, dataset_data, es_data, en_data)
